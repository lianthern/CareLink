"""Capture this bounded inventory into verified, encrypted GitHub Release capsules."""
import hashlib
import json
import os
import subprocess
import time
import zipfile
from concurrent.futures import ThreadPoolExecutor
from collections import deque
from pathlib import Path

os.umask(0o077)
ROOT=Path('/tmp/unseen-archive-20260920')
KEY=Path('/Users/max/.config/unseen-archive/owner-20260920.agekey')
REPO='maxi-max-dev/unseen'
TAG='unseen-archive-20260920'
PART=64*1024*1024
PACK=256*1024*1024
recipient=subprocess.check_output(['age-keygen','-y',str(KEY)],text=True).strip()

def run(args, **kw):
    return subprocess.run(args,check=True,**kw)

def atomic_json(path,value):
    temporary=path.with_suffix(path.suffix+'.new')
    temporary.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
    temporary.replace(path)

def digest(path):
    with path.open('rb') as stream:return hashlib.file_digest(stream,'sha256').hexdigest()

def release():
    return json.loads(subprocess.check_output(['gh','release','view',TAG,'--repo',REPO,'--json','id,assets,isDraft,url'],text=True))

try:
    info=release()
except subprocess.CalledProcessError:
    notes=ROOT/'release-notes-draft.md'
    notes.write_text('UNSEEN 项目档案快照正在构建。附件原件使用 age 加密，索引将在校验后发布。此快照不是生产部署，也不把占位文件算作已备份。\n')
    run(['gh','release','create',TAG,'--repo',REPO,'--target','6233df8534efc35b782bca5d0d84b5d2151f7500','--draft','--prerelease','--latest=false','--title','UNSEEN 项目档案 · 2026-09-20','--notes-file',str(notes)])
    info=release()
if not info['isDraft']:raise RuntimeError('Refusing to append to a published archive release')

packs_path=ROOT/'packs.private.json'
objects_path=ROOT/'objects.private.json'
packs=json.loads(packs_path.read_text()) if packs_path.exists() else []
objects=json.loads(objects_path.read_text()) if objects_path.exists() else {}
rows=[json.loads(line) for line in (ROOT/'files.private.jsonl').read_text().splitlines()]
extra=[]

# Save every available Git ref/reflog into portable bundles, without changing a source worktree.
histories_path=ROOT/'histories.private.json'
histories=json.loads(histories_path.read_text()) if histories_path.exists() else []
known_history={row['commonDir'] for row in histories}
for i,repo in enumerate(json.loads((ROOT/'git-repositories.private.json').read_text())):
    if repo['commonDir'] in known_history:continue
    bundle=ROOT/f'history-{i:03d}.bundle'
    refs=subprocess.check_output(['git','--git-dir',repo['commonDir'],'show-ref','--head'],text=True).splitlines()
    if not refs:
        histories.append({**repo,'status':'no-commits'});continue
    if not bundle.exists():
        with (ROOT/f'history-{i:03d}.log').open('wb') as log:
            run(['git','--git-dir',repo['commonDir'],'bundle','create',str(bundle),'--all','--reflog','HEAD'],stdout=log,stderr=log)
            run(['git','--git-dir',repo['commonDir'],'bundle','verify',str(bundle)],stdout=log,stderr=log)
    value={'source':'git-history','path':bundle.name,'localPath':str(bundle),'id':hashlib.sha256(repo['commonDir'].encode()).hexdigest()[:24],
        'kind':'file','sha256':digest(bundle),'bytes':bundle.stat().st_size,'mtime':bundle.stat().st_mtime,'mode':0o600}
    histories.append({**repo,'status':'bundle-verified','file':value,'refs':refs})
    atomic_json(histories_path,histories)
    print(json.dumps({'event':'git_bundle_verified','repository':i,'bytes':value['bytes']}),flush=True)
extra.extend(row['file'] for row in histories if row.get('file'))

# Keep the frozen original manifest; add separately captured staged/unstaged patches.
extra=json.loads((ROOT/'extra-files.private.json').read_text())
extra.extend(json.loads((ROOT/'supplemental-files.private.json').read_text()))

unique={}
for row in rows+extra:
    if row['kind']=='file':unique.setdefault(row['sha256'],row)
# Only skip complete objects whose every capsule has a verified upload record.
objects={sha:value for sha,value in objects.items() if all(part['pack']<len(packs) for part in value['parts'])}
pending=[row for sha,row in sorted(unique.items()) if sha not in objects]
print(json.dumps({'event':'archive_started','uniqueFiles':len(unique),'remainingFiles':len(pending),'alreadyUploadedPacks':len(packs)}),flush=True)

archive=None; plain=None; pack_bytes=0; pack_members=0
next_index=len(packs)
pool=ThreadPoolExecutor(max_workers=3)
inflight=deque()

def open_pack():
    global archive,plain,pack_bytes,pack_members
    plain=ROOT/f'working-pack-{next_index:04d}.zip'
    if plain.exists():raise RuntimeError(f'Unfinished local capsule must be reviewed: {plain.name}')
    archive=zipfile.ZipFile(plain,'w',compression=zipfile.ZIP_DEFLATED,compresslevel=1,allowZip64=True)
    pack_bytes=0;pack_members=0

def upload_pack(index,plain,encrypted,plain_sha,members,uncompressed):
    encrypted_sha=digest(encrypted)
    asset_name=f'capsule-{index:04d}-{encrypted_sha[:16]}.zip.age'
    ready=ROOT/asset_name;encrypted.rename(ready)
    size=ready.stat().st_size
    match=None
    for attempt in range(5):
        existing=next((a for a in release()['assets'] if a['name']==asset_name),None)
        if existing is None:
            result=subprocess.run(['gh','release','upload',TAG,str(ready),'--repo',REPO],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
            if result.returncode and attempt<4:time.sleep(2**(attempt+1));continue
        candidate=next((a for a in release()['assets'] if a['name']==asset_name),None)
        if candidate:
            remote=json.loads(subprocess.check_output(['gh','api',f"repos/{REPO}/releases/assets/{candidate['apiUrl'].split('/')[-1]}"],text=True))
            if remote.get('size')==size and remote.get('state')=='uploaded' and remote.get('digest')=='sha256:'+encrypted_sha:
                match=remote;break
        if attempt<4:time.sleep(2**(attempt+1))
    if match is None:raise RuntimeError(f'No matching remote SHA256 receipt for {asset_name}')
    record={'asset':asset_name,'bytes':size,'sha256':encrypted_sha,'plaintextSha256':plain_sha,
        'members':members,'uncompressedBytes':uncompressed,'assetId':match['id'],'url':match['browser_download_url']}
    return index,record,plain,ready

def drain():
    future=inflight.popleft()
    index,record,plain,ready=future.result()
    if index!=len(packs):raise RuntimeError('Upload sequence mismatch')
    packs.append(record)
    atomic_json(packs_path,packs)
    atomic_json(objects_path,objects)
    print(json.dumps({'event':'capsule_uploaded_verified','capsule':len(packs),'bytes':record['bytes'],'filesSoFar':len(objects),'totalUploadedBytes':sum(p['bytes'] for p in packs)}),flush=True)
    plain.unlink();ready.unlink()

def finish_pack():
    global archive,plain,next_index
    if archive is None:return
    archive.close();archive=None
    plain_sha=digest(plain)
    encrypted=plain.with_suffix('.zip.age.partial')
    run(['age','-r',recipient,'-o',str(encrypted),str(plain)])
    child=subprocess.Popen(['age','-d','-i',str(KEY),str(encrypted)],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    recovered=hashlib.file_digest(child.stdout,'sha256').hexdigest()
    if child.wait()!=0 or recovered!=plain_sha:raise RuntimeError('Encryption round-trip failed')
    inflight.append(pool.submit(upload_pack,next_index,plain,encrypted,plain_sha,pack_members,pack_bytes))
    next_index+=1
    plain=None
    if len(inflight)>=3:drain()

for n,row in enumerate(pending,1):
    parts=[];file_hash=hashlib.sha256();remaining=row['bytes'];index=0
    source=Path(row['localPath'])
    with source.open('rb') as stream:
        while remaining or (row['bytes']==0 and not parts):
            length=min(PART,remaining)
            if archive is not None and pack_bytes+length>PACK:finish_pack()
            if archive is None:open_pack()
            member=f"objects/{row['sha256']}/{index:05d}"
            chunk_hash=hashlib.sha256()
            entry=zipfile.ZipInfo(member,date_time=(2026,9,20,0,0,0));entry.compress_type=zipfile.ZIP_DEFLATED;entry.external_attr=0o600<<16
            with archive.open(entry,'w',force_zip64=True) as output:
                needed=length
                while needed:
                    data=stream.read(min(1024*1024,needed))
                    if not data:raise RuntimeError('Source shortened during archive')
                    output.write(data);file_hash.update(data);chunk_hash.update(data);needed-=len(data)
            parts.append({'pack':next_index,'member':member,'bytes':length,'sha256':chunk_hash.hexdigest()})
            pack_bytes+=length;pack_members+=1;remaining-=length;index+=1
        if stream.read(1):raise RuntimeError('Source grew during archive')
    if file_hash.hexdigest()!=row['sha256']:raise RuntimeError(f"Source changed during archive: {row['id']}")
    objects[row['sha256']]={'bytes':row['bytes'],'parts':parts}
    if n%1000==0:print(json.dumps({'event':'files_captured','files':n,'remaining':len(pending)-n}),flush=True)
finish_pack()
while inflight:drain()
pool.shutdown(wait=True)
atomic_json(objects_path,objects)
atomic_json(ROOT/'upload-complete.json',{'schema':'unseen-capture-completion/1','repository':REPO,'tag':TAG,
    'recipient':recipient,'packs':len(packs),'objects':len(objects),'uploadedBytes':sum(p['bytes'] for p in packs)})
print(json.dumps({'event':'archive_upload_complete','packs':len(packs),'objects':len(objects),'uploadedBytes':sum(p['bytes'] for p in packs)}),flush=True)
