import concurrent.futures,hashlib,json,os,stat,time
from pathlib import Path
os.umask(0o077)
r=Path('/tmp/unseen-archive-20260920')
sources=json.loads((r/'sources.private.json').read_text());candidates=json.loads((r/'additional-candidates.private.json').read_text());base_names=['/Users/max/Library/Mobile Documents/com~apple~CloudDocs','/Users/max/Documents','/Users/max/Desktop','/Users/max/Downloads']
base_names=[b for b in base_names if any(x['path'].startswith(b+'/') for x in candidates)]
mapping={}
for base in base_names:
 sid=f's{len(sources)+1:03d}'
 source={'id':sid,'path':base,'label':str(Path(base).relative_to('/Users/max'))+' [selected UNSEEN files]','selection':'Only project-named paths discovered by filename inventory; unrelated files excluded'}
 sources.append(source);mapping[base]=source

def capture(x):
 p=Path(x['path']);base=next(b for b in base_names if str(p).startswith(b+'/'));source=mapping[base];rel=str(p.relative_to(base));sid=source['id']
 row={'source':sid,'path':rel,'localPath':str(p),'id':hashlib.sha256((sid+'\0'+rel).encode()).hexdigest()[:24]}
 try:
  before=p.lstat();row.update(bytes=before.st_size,mtime=before.st_mtime,mode=stat.S_IMODE(before.st_mode))
  if stat.S_ISLNK(before.st_mode):row.update(kind='symlink',target=os.readlink(p));return row
  if not stat.S_ISREG(before.st_mode):row['kind']='special-file';return row
  if before.st_flags & stat.SF_DATALESS:row['kind']='cloud-placeholder';return row
  with p.open('rb') as f:row['sha256']=hashlib.file_digest(f,'sha256').hexdigest()
  after=p.stat()
  row['kind']='file' if (before.st_size,before.st_mtime_ns)==(after.st_size,after.st_mtime_ns) else 'changed-during-scan'
 except OSError as e:row.update(kind='unreadable',error=type(e).__name__)
 return row
rows=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
 for x in pool.map(capture,candidates):
  rows.append(x)
  if len(rows)%250==0:print(json.dumps({'event':'additional_hashed','files':len(rows),'total':len(candidates)}),flush=True)
private=r/'additional-capture-manifest.json';private.write_text(json.dumps({'sources':list(mapping.values()),'files':rows},ensure_ascii=False,indent=2)+'\n')
(r/'additional-files.private.json').write_text(json.dumps(rows,ensure_ascii=False,indent=2)+'\n')
(r/'sources.private.json').write_text(json.dumps(sources,ensure_ascii=False,indent=2)+'\n')
with (r/'files.private.jsonl').open('a') as out:
 for x in rows:out.write(json.dumps(x,ensure_ascii=False)+'\n')
supplemental=json.loads((r/'supplemental-files.private.json').read_text());supplemental.append({'source':'capture-metadata','path':private.name,'localPath':str(private),'id':'additional-capture-20260920','bytes':private.stat().st_size,'mtime':private.stat().st_mtime,'mode':0o600,'kind':'file','sha256':hashlib.sha256(private.read_bytes()).hexdigest()})
(r/'supplemental-files.private.json').write_text(json.dumps(supplemental,ensure_ascii=False,indent=2)+'\n')
from collections import Counter
print(json.dumps({'event':'additional_capture_complete','files':len(rows),'sources':len(mapping),'kinds':dict(Counter(x['kind'] for x in rows))}),flush=True)
