from pathlib import Path
import json,hashlib,stat,os,collections
r=Path('/tmp/unseen-archive-20260920')
rows=[json.loads(l) for l in (r/'files.private.jsonl').read_text().splitlines()]
known={x['localPath'] for x in rows}
for entry in json.loads((r/'cloud-git-candidates.private.json').read_text()):
 if entry['path'] in known:continue
 p=Path(entry['path']);rel=str(p.relative_to('/Users/max/Library/Mobile Documents/com~apple~CloudDocs'));st=p.lstat()
 rows.append({'source':'s066','path':rel,'localPath':str(p),'id':hashlib.sha256(('s066\0'+rel).encode()).hexdigest()[:24],'bytes':st.st_size,'mtime':st.st_mtime,'mode':stat.S_IMODE(st.st_mode),'kind':'cloud-placeholder'})
updated=0
for row in rows:
 if row['kind'] not in ('cloud-placeholder','unreadable','changed-during-scan'):continue
 p=Path(row['localPath'])
 try:
  before=p.stat()
  if before.st_flags & stat.SF_DATALESS:continue
  with p.open('rb') as f:sha=hashlib.file_digest(f,'sha256').hexdigest()
  after=p.stat()
  if (before.st_size,before.st_mtime_ns)!=(after.st_size,after.st_mtime_ns):continue
  row.update(kind='file',sha256=sha,bytes=after.st_size,mtime=after.st_mtime,mode=stat.S_IMODE(after.st_mode));row.pop('error',None);updated+=1
 except OSError:continue
p=r/'files.private.jsonl.new'
with p.open('w') as out:
 for row in rows:out.write(json.dumps(row,ensure_ascii=False)+'\n')
p.replace(r/'files.private.jsonl')
print(json.dumps({'updated':updated,'rows':len(rows),'kinds':dict(collections.Counter(x['kind'] for x in rows))}),flush=True)
