def inspect(item):
    source,location=item;p=Path(location)
    row={'source':source['id'],'path':str(p.relative_to(source['path'])),'localPath':location}
    row['id']=hashlib.sha256((source['id']+'\0'+row['path']).encode()).hexdigest()[:24]
    try:
        before=p.lstat();row.update(bytes=before.st_size,mtime=before.st_mtime)
        if stat.S_ISLNK(before.st_mode):row.update(kind='symlink',target=os.readlink(p));return row
        if not stat.S_ISREG(before.st_mode):row['kind']='special-file';return row
        if before.st_flags & stat.SF_DATALESS:row['kind']='cloud-placeholder';return row
        row['kind']='file'
        with p.open('rb') as f:row['sha256']=hashlib.file_digest(f,'sha256').hexdigest()
        after=p.stat()
        if (before.st_size,before.st_mtime_ns,before.st_ino)!=(after.st_size,after.st_mtime_ns,after.st_ino):row['kind']='changed-during-scan'
        row['mode']=stat.S_IMODE(before.st_mode)
    except OSError as e:row.update(kind='unreadable',error=e.__class__.__name__)
    return row
